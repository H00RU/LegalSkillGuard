Quotes must be exact and inside cited current units. Do not create answers or use current hypotheses in the source decoder. Read the actual complete source.

Additional requirements are typed evidence holes that are required to close a claim even if they are not arguments of its term. A claim can reuse the same provider as another claim. Legal actors/actions/triggers/conditions/exceptions are decoded from actual words, not supplied by host rules.
Keep a genuine distinction between local observations and nonlocal binding: where a normative header supplies an actor/modality or another clause supplies a condition, emit source-grounded values there and a typed hole/requirement on the action claim. Do not resolve all nonlocal roles inside the decoder or merely reproduce closed existing norms; the experiment needs actual binding examples. Equally, do not manufacture ambiguity or strip a genuinely local explicit qualification to give the learner work. Preserve real actor/owner, trigger and exception scope. Quotes identify actual fragments generated by your decoder; a selector's quote can be a unique substring of that fragment's grounded quote. def bundle(folder):
    proposal=read(folder/'frontend_teacher.json')['value'];jsonschema.validate(proposal,FRONT)
    base=read(folder/'base.json');spec=deepcopy(base['grammar']);spec['constructors']+=proposal['extra_constructors']
    constructors(spec)
    parts=[]
    for value in proposal['source_functions']:
        tree=ast.parse(value)
        if any(not isinstance(n,ast.FunctionDef) for n in tree.body):raise ValueError('Only pure function definitions')
        parts.extend(ast.get_source_segment(value,n) for n in tree.body)
    code=function_update(base['source_program'],parts)
    if any(isinstance(n,ast.Constant) and n.value in ('Entailment','Contradiction','NotMentioned') for n in ast.walk(ast.parse(code))):
        raise ValueError('Native labels in a source decoder')
    value={'spec':spec,'program':code,'binding_examples':proposal['binding_examples']}
    immutable_json(folder/'frontend_compiled.json',value)
    return value


def decode(b,source):
    pub=source['views'][0]['public'];spans=source_spans(material_public(pub));units,supports=execution_source_units(pub['source_text'],spans)
    raw=execute_json(b['program'],{'source_text':pub['source_text'],'source_spans':units},{},allow_helpers=True)
    jsonschema.validate(raw,OUTPUT);schema=constructors(b['spec']);fragments=[];last_header=None
    unit_info={}
    for i,u in enumerate(units):
        if u['text'].rstrip().endswith(':'):last_header=i
        unit_info[u['id']]={'ordinal':i,'header':last_header}
    for fragment in raw['fragments']:
        ids=expand_unit_support(fragment['support_ids'],supports)
        a=min(spans[i]['start'] for i in ids);z=max(spans[i]['end'] for i in ids)
        hits=occurrences(pub['source_text'][a:z],fragment['quote'])
        if len(hits)!=1:raise ValueError('Fragment quote must ground uniquely in actual cited source')
        begin,end=hits[0];begin+=a;end+=a
        term=fragment['term'];req=fragment['requires']
        anchor=min((unit_info[i] for i in fragment['support_ids']),key=lambda u:u['ordinal'])
        fragments.append({'id':len(fragments),'kind':fragment['kind'],'quote':fragment['quote'],
            'start':begin,'end':end,'unit':anchor['ordinal'],'header':anchor['header'],
            'term':term,'requires':req,'support_ids':ids})
    validate_fragments(fragments,schema)
    return {'source':source['key'],'fragments':fragments,'decoder_unresolved':raw['unresolved'],'raw_output':raw}


def selector(fragment,choice,text):
    return fragment['term']==json.loads(choice['term_json']) and bool(occurrences(text[fragment['start']:fragment['end']],choice['quote']))


def fit(root):
    folder,p,ledger,public=setup(root);b=bundle(folder);cases=[]
    for key in p['teacher_sources']:
        case=decode(b,public[key]);immutable_json(folder/'training_fragments'/(key.replace(':','_')+'.json'),case)
        case['bindings']=[];text=public[key]['views'][0]['public']['source_text']
        for example in b['binding_examples']:
            if example['source']!=key:continue
            claims=[f for f in case['fragments'] if f['kind']=='claim' and selector(f,example['claim'],text)]
            if len(claims)!=1:raise ValueError('Acquired claim selector is absent or ambiguous')
            providers={}
            for item in example['providers']:
                found=[f for f in case['fragments'] if f['kind']=='value' and selector(f,item,text)]
                if len(found)!=1 or item['slot'] in providers:raise ValueError('Acquired provider is absent, ambiguous or duplicated')
                providers[item['slot']]=found[0]['id']
            case['bindings'].append({'claim':claims[0]['id'],'providers':providers,'positive':example['positive']})
        cases.append(case)
    if any(d['source'] not in p['teacher_sources'] for d in b['binding_examples']):raise ValueError('Non-TRAIN demonstration')
    schema=constructors(b['spec']);models={'Joint':learn(cases,schema),'UnaryRelations':learn(cases,schema,cross_slot=False)}
    for arm,model in models.items():immutable_json(folder/(arm+'_model.json'),model)
    immutable_json(folder/'training_cases.json',{'cases':cases,'models':models,'goal_complete':False})
    print(json.dumps({a:{'policies':len(m['policies']),'rejected':len(m['rejected_families'])} for a,m in models.items()}),flush=True)
    return b,cases,models

Inputs={source_text,source_spans:common current units}; premises={fragments:actual acquired current fragments}. Return {statements:[{term:closed acquired constructor list,support_ids:[CURRENT UNIT IDs]}],unresolved:[string]}. You can choose an ordinary context/relational policy, any internal decomposition, independently learn a joint binding policy, or directly parse full original source. You are not restricted to nearest matching or to our model's rule language. All source labels, grounded binding examples, common decoder, ontology and consumer are available. The actual learned Joint policies and fixed reuse material are not supplied. No native answer labels/source answer tables or company/source applicability keys. Each statement needs actual current source support. Keep unresolved evidence explicit. Return rationale and complete program.'''
    task={'training_material':packed_training(root,p,public),'common_decoder':b,
        'actual_training_fragments_and_binding_supervision':cases}
    wire,host=decoder_schema(FREE);request={'system':system,'user':pack(task),'wire_schema':wire,'logical_schema':FREE,'host_constraints':host}

def evaluate(root,reuse=False):
    folder,p,ledger,public=setup(root)
    if any(v.get('reserved',0) for v in physical_snapshot(ledger.path).values()):raise ValueError('Existing request still live')
    b,cases,models=fit(root);free=read(folder/'free_teacher.json')['value'];jsonschema.validate(free,FREE)
    validate_procedure(free['program'],allow_helpers=True)
    if reuse:
        if not (folder/'development/completed.json').exists():raise ValueError('Complete development first')
        immutable_json(folder/'reuse_freeze.json',{'models':models,'free':free,'frontend_digest':digest(b),
            'reuse_sources':p['reuse_sources'],'no_adaptive_reuse_repair':True})
    stage='reuse' if reuse else 'development';keys=p['reuse_sources'] if reuse else p['development_sources']
    grading=read(root/'artifacts/legal_skill_evolution_20261111/prepared/train/grading.json');base=warm_base(root,p);schema=constructors(b['spec'])
    results={a:[] for a in ('Joint','UnaryRelations','NearestContext','FreeProgram')}
    for key in keys:
        source=public[key];case=decode(b,source);immutable_json(folder/stage/'fragments'/(key.replace(':','_')+'.json'),case)
        spans=source_spans(material_public(source['views'][0]['public']));units,supports=execution_source_units(source['views'][0]['public']['source_text'],spans)
        for arm in results:
            path=folder/stage/arm/(key.replace(':','_')+'.json')
            if path.exists():results[arm].append(read(path));continue
            if arm=='FreeProgram':
                out=execute_json(free['program'],{'source_text':source['views'][0]['public']['source_text'],'source_spans':units},{'fragments':case['fragments']},allow_helpers=True)
                jsonschema.validate(out,RESOLVED)
                for n in out['statements']:n['support_ids']=expand_unit_support(n['support_ids'],supports)
            else:out=resolve(models.get(arm,models['Joint']),case['fragments'],schema,nearest=arm=='NearestContext')
            nodes=[]
            for n in out['statements']:
                sort=term_sort(n['term'],schema)
                if sort not in ('Norm','Definition','Description'):continue
                nodes.append({'id':len(nodes),'sort':sort,'term':n['term'],'support_ids':n['support_ids'],'child_ids':[],'ambiguous':False})
            interp={'nodes':nodes,'roots':[n['id'] for n in nodes]};history={v['view']:v for v in base[key]['views']};views=[]
            for v in source['views']:
                execution=apply_current(b['spec'],{'rules':[],'disabled_rules':[],'acquired_pairs':[]},interp,v);x=execution['value'];handled=x['handled'] and not case['decoder_unresolved'] and not out['unresolved']
                ids=sorted({s for i in x['used_nodes'] for s in nodes[i]['support_ids']})
                proof=sorted({spans[i]['public_sentence_id'] for i in ids if 'public_sentence_id' in spans[i]})
                answer=x['answer'] if handled else history[v['id']]['execution']['answer']
                score=grade_view('ContractNLI',v,grading[key][v['id']],answer,proof) if handled else history[v['id']]['score']
                views.append({'view':v['id'],'core_executed':handled,'fallback':not handled,'answer':answer,'score':score,'proof_sentence_ids':proof,'execution':execution,'source_unresolved_gate':bool(case['decoder_unresolved'] or out['unresolved'])})
            row={'source':key,'arm':arm,'interpretation':interp,'binding':out,'decoder_unresolved':case['decoder_unresolved'],'views':views,'source_semantic_review_pending':True,'goal_complete':False}
            immutable_json(path,row);results[arm].append(row)
            print(json.dumps({'stage':stage,'arm':arm,'source':key,'core':sum(v['core_executed'] for v in views),'native_correct':sum(v['core_executed'] and v['score']['correct'] for v in views)}),flush=True)
    immutable_json(folder/stage/'completed.json',{'results':results,'ledger':physical_snapshot(ledger.path),'goal_complete':False})
